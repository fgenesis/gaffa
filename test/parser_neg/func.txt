-- Malformed function definitions
### mutual recursion without defer
func aa() return bb()
func bb() return aa()
### function redefinition
func f() {}
func f() {}
### function name clashes with a variable
uint f = 1
func f() {}
### variadic parameter not last
func f(uint ..., uint x) {}
### two variadic parameters
func f(uint ..., float ...) {}
### variadic return type not last
func f() -> uint..., float {}
### '-> ?' was removed, '-> ...' means the same
func f() -> ? {}
### missing parameter list
func f {}
### missing body
func f()
### anonymous function as a statement
func() {}
### method definition with :::
type T = {x=float}
func T:::len() return 1
### namespaced function defined with ::
type T = {x=float}
func T::f() {}
### return arrow without types
func f() -> {}
### unknown function attribute
func [fast] f() {}### less-than can't be overloaded
func int.<(int a, int b) -> bool { return true }
### greater-or-equal can't be overloaded
func int.>=(int a, int b) -> bool { return true }
### not-equal can't be overloaded
func int.!=(int a, int b) -> bool { return true }
### logical and can't be overloaded
func bool.&&(bool a, bool b) -> bool { return a }
### prefix-only operator without the prefix name
func int.!(int a) -> bool { return true }
### unwrap can't be overloaded
func int.prefix*(int a) -> int { return a }
### prefix name with a space
func int.prefix -(int a) -> int { return a }
### string literal as function name
func T."x"() {}
### operator name with a space inside
func T:[ ](uint k) -> uint { return k }
### keyword as function name
func T.if() {}
### operator name at end of file
func T.+
