func T.if() {}

uint a = 0x

uint a = 0xfg

uint a = 0b102

uint a = 0o9
### exponent without digits
float a = 1e

float a = 1e+

uint a = 18446744073709551616

string s = "\q"

string s = "\u{}"

string s = "\u{41"
