-- Exports and imports: syntax that must parse (docs/agents/language.md, "Exports and imports")
### export prefix at top level
export uint a = 1
export func f() -> uint { return 1 }
### export prefix in a nested block
func M() -> symtab
{
    if(true) { export func f() -> uint { return 1 } }
    return export
}
### named export
func M() -> symtab
{
    export ("g") func f() -> uint { return 1 }
    return export
}
### exported method and namespaced symbol
func M(type T) -> symtab, type
{
    export type B = { v = T }
    export func B:get() -> T { return self.v }
    export B::zero = 0
    return export, B
}
### export as expression
func M() -> symtab
{
    var e = export
    return e
}
### export field assignment
func M() -> symtab
{
    export.hello = func() -> uint { return 1 }
    export["g"] = 5
    return export
}
### symtab methods on export
func M() -> symtab
{
    export:merge(N())
    return export
}
### import prefix
type X = import M(float)
### import prefix with several values
var a, b, ___ = import M()
### import statement
import M()
### import in a nested block
{ import M() }
### import inside call arguments
g(import M())
### import in the middle of a value list
var a, b = 1, import M()
### import of a symtab local
var s, type X = M(float)
import s
### import binds like a unary operator
uint x = import M() + 1
### import with value count adjust
uint a = import V()#2
### comptime
uint x = comptime f(2)
var a, b = comptime g()
### __exports used by name
func M() -> symtab
{
    export uint a = 1
    return __exports
}
### __exports as an upvalue
func M() -> symtab
{
    export uint a = 1
    var g = func() -> symtab { return __exports }
    return g()
}
