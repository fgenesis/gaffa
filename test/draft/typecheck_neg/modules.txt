-- Exports and imports that must fail to link/fold
### symtab not taken: unbalanced declaration
## error: unbalanced declaration
func [pure] Box(type T) -> symtab, type
{
    export type B = { v = T }
    return export, B
}
type FB = Box(float)
### export = import with a single symtab
## error: unbalanced; did you mean 'export import f()'?
func [pure] Inner() -> symtab
{
    export func f() -> uint { return 1 }
    return export
}
func [pure] Outer() -> symtab
{
    export = import Inner()
    return export
}
### import of a value that doesn't fold
## error: import needs a constant symtab
func M(uint n) -> symtab
{
    export uint f = n
    return export
}
uint n := 1
n = 2
import M(n)
### import target given a non-symtab (old return order)
## error: import needs a symtab
func [pure] Box(type T) -> type, symtab
{
    export type B = { v = T }
    return B, export
}
type FB = import Box(float)
### export target given a non-symtab
## error: export target needs a symtab
func M() -> symtab
{
    export = 5
    return export
}
### methods not imported
## error: no method get
func [pure] Box(type T) -> symtab, type
{
    export type B = { v = T }
    export func B:get() -> T { return self.v }
    return export, B
}
var _, type FB = Box(float)
FB b = {v = 1.0}
float x = b:get()
### methods don't follow values
## error: no method get
func [pure] Box(type T) -> symtab, type
{
    export type B = { v = T }
    export func B:get() -> T { return self.v }
    return export, B
}
func mk()
{
    var _, type FB = Box(float)
    return FB{v = 1.0}
}
var b = mk()
float x = b:get()

### export target clashes with an existing export
## error: already exported

{

}

{
    export func f() -> uint { return 2 }

    return export
}
