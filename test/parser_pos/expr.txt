-- Expressions
### arithmetic precedence
uint a = 1 + 2 * 3 - 4 // 2
float b = 1.0 / 2.0 + 3.0
### modulo
uint a = 7 % 3
### bitwise operators
uint a = 1 & 3
uint b = 1 | 2
uint c = 1 ^ 3
uint d = 1 << 4
uint e = 256 >> 2
uint f = ~0
### bitwise binds tighter than comparison
bool a = 1 & 3 == 1
### logic operators
bool a = true && false || true
bool b = !true
### and/or keywords
var a = nil or 5
var b = 1 and 2
### comparison operators
bool a = 1 < 2
bool b = 1 <= 2
bool c = 1 > 2
bool d = 1 >= 2
bool e = 1 == 2
bool f = 1 != 2
### string concatenation
string s = "a" ++ "b" ++ "c"
### unary operators
int a = -(1)
int b = - -1
uint c = #"abc"
bool d = !!true
uint e = ~~1
### unary minus on a variable
uint u = 5
int i = -u
### length of a parenthesized expression
uint n = #("a" ++ "b")
### unwrap operator
uint? o = 1
uint v = *o
### capture operator
var r = ??(1 // 0)
### index and field access
table t = {x=1, y={z=2}}
var a = t.x
var b = t["x"]
var c = t.y.z
var d = t["y"]["z"]
### field and index after a call
func f() return {x=1}
var a = f().x
var b = f()["x"]
### field and index after a parenthesized expression
table t = {x=1}
var a = (t).x
var b = (t)["x"]
### index after an array constructor
var a = [1, 2, 3][0]
### method calls
string s = "abc"
var a = s:len()
var b = s:["len"]()
var c = s:sub(0, 1):len()
### method call on a literal
var n = "abc":len()
### namespace lookup
var a = uint::maxval
var b = string::["len"]
### chained calls
func mk() return func(uint x) return x
var a = mk()(1)
### trailing comma in call arguments
func f(uint a, uint b) return a
var x = f(1, 2,)
### call with adjusted value count
func three() return 1, 2, 3
var a, b = three()#2
### grouping
uint a = ((((1))))
uint b = (1 + 2) * (3 + 4)
### table constructors
table a = {}
table b = {1, 2, 3}
table c = {x=1, y=2}
table d = {1, x=2, 3}
table e = {[1+1]=2, ["k"]=3}
table f = {1; 2; 3}
table g = {1, 2, 3,}
table h = {{}, {{}}}
### array constructors
var a = [1, 2, 3]
var b = []
var c = [1, 2, 3,]
var d = [[1], [2, 3]]
### closures as values
var f = func(uint x) return x
var g = func(uint x) { return x }
var h = func(uint x) -> uint return x
### closure called immediately
uint a = (func(uint x) return x)(1)
### unwrap binding in condition
uint? o = 1
if(o => v) { uint w = v }
### unwrap binding combined with another condition
uint? o = 1
if(o => v && v > 0) { uint w = v }
