-- Declarations
### C-style declarations
int a = +1
uint b = 2
float c = 3.0
string d = "four"
### var with deduced type
var a = 1
var b = -1
var c = 1.0
var d = "s"
### var with explicit type
var a: int = +1
var b: uint = 2
### var with type expression
var t: Table(uint, uint) = {}
### mixed declaration lists
uint x, string y = 0, "hi"
uint a, b, float c = 0, 1, 0.5
uint p, var q, r, float s = 0, 1, "hi", 0.5
var m:uint, n:uint = 0, 1
### mutable declarations
uint a := 0
var b := 1
uint c, d := 1, 2
a = 5
b = 6
c, d = d, c
### optional type C-style
int? x = nil
### optional type after var
var x: int? = nil
### optional type later in a declaration list
uint a, int? b = 1, nil
### optional parameter type
func f(int? x) return x
### sink in declaration list
uint a, _ = 1, 2
### sink as first name
var _, b = 1, 2
### shadowing in a nested block
uint x = 1
{
    var x = x
    uint y = x
}
### same name in sibling blocks
{ uint x = 1 }
{ uint x = 2 }
### declarations separated by semicolons
uint a = 1; uint b = 2;
### several declarations on one line
uint a = 1 uint b = 2
### deferred declaration
int x = defer
func f() return x
int x = +0
### deferred declaration re-declared mutable
int x = defer
func f() x = +1
int x := +0
### deferred declaration of a function
var f = defer
func g() return f()
func f() return 1
### type declarations
type T = uint
type P = {x=float, y=float}
type U := uint
